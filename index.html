function showVK(m, generation) { const container = $('vk'); container.replaceChildren(); container.style.display = 'block'; const frame = document.createElement('iframe'); frame.id = 'vkframe'; const url = new URL(m.embed); url.hostname = 'vk.com'; url.searchParams.set('js_api', '1'); frame.src = url.href; frame.style.width = '100%'; frame.style.height = '100%'; frame.style.border = '0'; frame.allow = 'autoplay; encrypted-media; fullscreen; picture-in-picture'; frame.allowFullscreen = true; container.appendChild(frame); let initialized = false; function init() { if (initialized || generation !== playerGeneration) return; if (!window.VK?.VideoPlayer) return; try { const player = VK.VideoPlayer(frame); vkPlayer = player; initialized = true; const events = VK.VideoPlayer.Events; player.on(events.inited, () => { if (generation !== playerGeneration) return; socket.emit('request-sync'); }); for (const event of [ events.started, events.resumed, events.paused ]) { player.on(event, () => { if (generation !== playerGeneration) return; if (Date.now() >= remoteUntil) { sendSync(); } }); } player.on(events.error, error => { console.warn('VK playback error:', error); }); } catch (error) { console.warn('VK initialization:', error); } } if (window.VK?.VideoPlayer) { init(); } else { let script = document.getElementById('vkapi'); if (!script) { script = document.createElement('script'); script.id = 'vkapi'; script.src = 'https://vk.com/js/api/videoplayer.js'; document.head.appendChild(script); } script.addEventListener('load', init, { once: true }); } }
